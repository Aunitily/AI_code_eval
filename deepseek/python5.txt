#!/usr/bin/env python3
"""
reg_gen.py - 从 Excel 寄存器描述生成 RTL、UVM 和 C 头文件

用法:
    python reg_gen.py registers.xlsx --out_dir ./generated --base_addr 0x4000_0000
"""

import argparse
import os
import sys
import math

try:
    import openpyxl
except ImportError:
    sys.exit("请安装 openpyxl: pip install openpyxl")

# ----------------------------------------------------------------------
# 解析 Excel
# ----------------------------------------------------------------------
def read_registers(xlsx_path):
    wb = openpyxl.load_workbook(xlsx_path, read_only=True)
    if 'registers' not in wb.sheetnames:
        raise SystemExit(f"未找到 Sheet 'registers'，可用 Sheet: {wb.sheetnames}")
    ws = wb['registers']
    rows = list(ws.iter_rows(values_only=True))
    if len(rows) < 2:
        raise SystemExit("Excel 文件没有数据行")

    # 表头
    header = [str(col).strip().lower() for col in rows[0]]
    expected_cols = ['register name', 'offset', 'field name', 'bit range', 'access', 'reset value']
    for col in expected_cols:
        if col not in header:
            raise SystemExit(f"缺少必要的列: {col}，实际列: {header}")

    regdata = []
    for row in rows[1:]:
        values = [str(c).strip() if c is not None else '' for c in row]
        entry = dict(zip(header, values))
        # 过滤完全空行
        if all(v == '' for v in entry.values()):
            continue
        regdata.append(entry)
    return regdata

# ----------------------------------------------------------------------
# 合并同一偏移量的字段为一个寄存器
# ----------------------------------------------------------------------
def group_registers(regdata):
    groups = {}   # key = offset (string like "0x00"), value = {name, fields[]}
    for entry in regdata:
        offset = entry['offset']
        name   = entry['register name']
        field  = {
            'name': entry['field name'],
            'bit_range': entry['bit range'],
            'access': entry['access'],
            'reset': entry['reset value'],
            'desc': entry.get('description', '')
        }
        if offset not in groups:
            groups[offset] = {'name': name, 'fields': []}
        # 注册器名必须一致
        if groups[offset]['name'] != name:
            raise SystemExit(f"偏移 {offset} 对应多个不同寄存器名: {groups[offset]['name']} 和 {name}")
        groups[offset]['fields'].append(field)
    return groups

# ----------------------------------------------------------------------
# 解析位范围 "2:1" 或 "0"，返回 (high, low)
# ----------------------------------------------------------------------
def parse_bit_range(brange):
    brange = brange.strip()
    if ':' in brange:
        high, low = brange.split(':')
        return int(high), int(low)
    else:
        bit = int(brange)
        return bit, bit

# ----------------------------------------------------------------------
# 转换为整数（支持 0x, 0b, 十进制）
# ----------------------------------------------------------------------
def parse_int(value):
    return int(value, 0)

# ----------------------------------------------------------------------
# 生成 Verilog RTL
# ----------------------------------------------------------------------
def gen_verilog(groups, out_dir, reg_width=32):
    """生成 regfile_top.v"""
    lines = []
    lines.append("// Auto-generated by reg_gen.py")
    lines.append("`timescale 1ns/1ps")
    lines.append("")
    lines.append(f"module regfile_top #(")
    lines.append(f"    parameter DATA_WIDTH = {reg_width}")
    lines.append(") (")
    lines.append("    input  wire                     clk,")
    lines.append("    input  wire                     rst_n,")
    lines.append("    input  wire [31:0]              addr_i,")   # 总线地址
    lines.append("    input  wire [DATA_WIDTH-1:0]    wr_data_i,")
    lines.append("    input  wire                     wr_en_i,")
    lines.append("    output reg  [DATA_WIDTH-1:0]    rd_data_o,")
    # 每个寄存器的值和写使能输出
    for offset, grp in sorted(groups.items(), key=lambda x: parse_int(x[0])):
        reg_name = grp['name']
        lines.append(f"    output wire [DATA_WIDTH-1:0]    {reg_name}_q,")
        lines.append(f"    output wire                     {reg_name}_we")
        if list(groups.keys())[-1] != offset:   # 最后加逗号，我们可以统一在例化时处理
            lines[-1] += ","
        else:
            lines[-1] += ""   # 最后不加逗号，下面结束括号
    lines.append(");")
    lines.append("")

    # 写使能解码
    for offset in sorted(groups.keys(), key=lambda x: parse_int(x)):
        off_val = parse_int(offset)
        reg_name = groups[offset]['name']
        lines.append(f"assign {reg_name}_we = (addr_i == 32'h{off_val:08X}) && wr_en_i;")
    lines.append("")

    # 寄存器实例化
    for offset in sorted(groups.keys(), key=lambda x: parse_int(x)):
        grp = groups[offset]
        reg_name = grp['name']
        # 计算复位值（聚合所有字段的复位值）
        reset_val = 0
        for field in grp['fields']:
            high, low = parse_bit_range(field['bit_range'])
            width = high - low + 1
            rst_val = parse_int(field['reset']) & ((1 << width) - 1)
            reset_val |= (rst_val << low)
        lines.append(f"// Register {reg_name} (offset 0x{parse_int(offset):X})")
        lines.append(f"reg [DATA_WIDTH-1:0] {reg_name}_reg;")
        lines.append(f"assign {reg_name}_q = {reg_name}_reg;")
        lines.append(f"always @(posedge clk or negedge rst_n) begin")
        lines.append(f"    if (!rst_n)")
        lines.append(f"        {reg_name}_reg <= {DATA_WIDTH}'h{reset_val:08X};")
        lines.append(f"    else if ({reg_name}_we)")
        lines.append(f"        {reg_name}_reg <= wr_data_i;")
        lines.append(f"end")
        lines.append("")

    # 读数据选择
    lines.append("// Read data mux")
    lines.append("always @(*) begin")
    lines.append("    case (addr_i)")
    for offset in sorted(groups.keys(), key=lambda x: parse_int(x)):
        off_val = parse_int(offset)
        reg_name = groups[offset]['name']
        lines.append(f"        32'h{off_val:08X}: rd_data_o = {reg_name}_q;")
    lines.append("        default: rd_data_o = '0;")
    lines.append("    endcase")
    lines.append("end")
    lines.append("")
    lines.append("endmodule")

    verilog_text = '\n'.join(lines)
    path = os.path.join(out_dir, 'regfile_top.v')
    with open(path, 'w') as f:
        f.write(verilog_text)
    print(f"生成 {path}")

# ----------------------------------------------------------------------
# 生成 UVM 模型
# ----------------------------------------------------------------------
def gen_uvm(groups, out_dir, base_addr=0, reg_width=32):
    """生成 reg_block.sv (包含寄存器类和 reg_block 类)"""
    lines = []
    lines.append("// Auto-generated by reg_gen.py")
    lines.append("`ifndef REG_BLOCK_SV")
    lines.append("`define REG_BLOCK_SV")
    lines.append("")
    lines.append("`include \"uvm_macros.svh\"")
    lines.append("import uvm_pkg::*;")
    lines.append("")

    # 生成每个寄存器的 UVM 类
    for offset in sorted(groups.keys(), key=lambda x: parse_int(x)):
        grp = groups[offset]
        reg_name = grp['name']
        class_name = f"{reg_name}_reg"
        lines.append(f"//------------------------------------------------------------------------")
        lines.append(f"// Register: {reg_name} (offset 0x{parse_int(offset):X})")
        lines.append(f"//------------------------------------------------------------------------")
        lines.append(f"class {class_name} extends uvm_reg;")
        lines.append(f"    `uvm_object_utils({class_name})")
        lines.append("")
        # 声明字段
        for field in grp['fields']:
            f_name = field['name']
            lines.append(f"    rand uvm_reg_field {f_name};")
        lines.append("")
        # build 方法
        lines.append(f"    function new(string name = \"{class_name}\");")
        lines.append(f"        super.new(name, {reg_width}, UVM_NO_COVERAGE);")
        lines.append(f"    endfunction")
        lines.append("")
        lines.append(f"    virtual function void build();")
        for field in grp['fields']:
            f_name = field['name']
            lines.append(f"        {f_name} = uvm_reg_field::type_id::create(\"{f_name}\");")
            high, low = parse_bit_range(field['bit_range'])
            width = high - low + 1
            access = field['access']   # 例如 "RW", "RO", "RC" 等
            rst_val = parse_int(field['reset']) & ((1 << width) - 1)
            lines.append(f"        {f_name}.configure(this, {width}, {low}, \"{access}\", 0, {rst_val}'h{rst_val:X}, 1, 1, 0);")
        lines.append(f"    endfunction")
        lines.append(f"endclass")
        lines.append("")

    # 生成 reg_block
    lines.append("//------------------------------------------------------------------------")
    lines.append("// Register Block")
    lines.append("//------------------------------------------------------------------------")
    lines.append(f"class reg_block extends uvm_reg_block;")
    lines.append(f"    `uvm_object_utils(reg_block)")
    lines.append("")
    # 声明寄存器实例
    for offset in sorted(groups.keys(), key=lambda x: parse_int(x)):
        reg_name = groups[offset]['name']
        class_name = f"{reg_name}_reg"
        lines.append(f"    rand {class_name} {reg_name};")
    lines.append("")
    lines.append(f"    function new(string name = \"reg_block\");")
    lines.append(f"        super.new(name, UVM_NO_COVERAGE);")
    lines.append(f"    endfunction")
    lines.append("")
    lines.append(f"    virtual function void build();")
    lines.append(f"        default_map = create_map(\"default_map\", 0, 4, UVM_LITTLE_ENDIAN);")
    for offset in sorted(groups.keys(), key=lambda x: parse_int(x)):
        reg_name = groups[offset]['name']
        class_name = f"{reg_name}_reg"
        off_val = parse_int(offset)
        # 基址 + 偏移
        abs_addr = base_addr + off_val
        lines.append(f"        {reg_name} = {class_name}::type_id::create(\"{reg_name}\");")
        lines.append(f"        {reg_name}.build();")
        lines.append(f"        default_map.add_reg({reg_name}, 'h{abs_addr:X}, \"RW\");")
    lines.append(f"        lock_model(); ")
    lines.append(f"    endfunction")
    lines.append(f"endclass")
    lines.append("")
    lines.append("`endif // REG_BLOCK_SV")

    uvm_text = '\n'.join(lines)
    path = os.path.join(out_dir, 'reg_block.sv')
    with open(path, 'w') as f:
        f.write(uvm_text)
    print(f"生成 {path}")

# ----------------------------------------------------------------------
# 生成 C 头文件
# ----------------------------------------------------------------------
def gen_c_header(groups, out_dir, base_addr=0):
    """生成 regs.h"""
    lines = []
    lines.append("// Auto-generated by reg_gen.py")
    lines.append("#ifndef REGS_H")
    lines.append("#define REGS_H")
    lines.append("")
    lines.append("#include <stdint.h>")
    lines.append("")
    lines.append(f"#define BASE_ADDR 0x{base_addr:08X}")
    lines.append("")

    for offset in sorted(groups.keys(), key=lambda x: parse_int(x)):
        grp = groups[offset]
        reg_name = grp['name']
        off_val = parse_int(offset)
        abs_addr = base_addr + off_val
        lines.append(f"// Register: {reg_name} (offset 0x{off_val:X})")
        lines.append(f"#define {reg_name}_ADDR (BASE_ADDR + 0x{off_val:X})")
        for field in grp['fields']:
            f_name = field['name']
            high, low = parse_bit_range(field['bit_range'])
            width = high - low + 1
            mask = ((1 << width) - 1) << low
            lines.append(f"#define {reg_name}_{f_name}_MASK  ((uint32_t)0x{mask:08X}U)")
            lines.append(f"#define {reg_name}_{f_name}_SHIFT {low}")
            lines.append("")
    lines.append("#endif // REGS_H")

    header_text = '\n'.join(lines)
    path = os.path.join(out_dir, 'regs.h')
    with open(path, 'w') as f:
        f.write(header_text)
    print(f"生成 {path}")

# ----------------------------------------------------------------------
# 主函数
# ----------------------------------------------------------------------
def main():
    parser = argparse.ArgumentParser(description='从 Excel 生成寄存器 RTL、UVM 和 C 头文件')
    parser.add_argument('xlsx', help='Excel 文件路径')
    parser.add_argument('--out_dir', default='./generated', help='输出目录 (默认 ./generated)')
    parser.add_argument('--base_addr', default='0x40000000', help='基址 (默认 0x40000000)')
    parser.add_argument('--reg_width', type=int, default=32, help='寄存器位宽 (默认 32)')
    args = parser.parse_args()

    base_addr = parse_int(args.base_addr)
    out_dir = args.out_dir
    os.makedirs(out_dir, exist_ok=True)

    # 1. 读取数据
    regdata = read_registers(args.xlsx)
    # 2. 按偏移量分组
    groups = group_registers(regdata)

    # 3. 生成文件
    gen_verilog(groups, out_dir, reg_width=args.reg_width)
    gen_uvm(groups, out_dir, base_addr=base_addr, reg_width=args.reg_width)
    gen_c_header(groups, out_dir, base_addr=base_addr)

    print("所有文件生成完成。")

if __name__ == '__main__':
    main()
	
脚本说明
输入输出

读取 Excel 文件的 registers 工作表，要求包含列：Register Name, Offset, Field Name, Bit Range, Access, Reset Value, Description（可选）。

生成以下 3 个文件：

regfile_top.v – 带总线接口的 Verilog 寄存器模块，包含所有寄存器及其输出、写使能逻辑。

reg_block.sv – UVM 寄存器模型（reg_block 类），含每个寄存器字段及各寄存器的地址映射。

regs.h – C 头文件，定义寄存器绝对地址宏与字段掩码/移位宏。

关键特性

多字段合并：相同偏移量的字段自动合并为一个 Verilog 寄存器和 UVM 寄存器类。

宽灵活度：复位值支持 0x、0b、十进制；访问类型（RW、RO、RC 等）直接传递给 UVM。

可配置性：通过命令行参数指定基址（--base_addr）、输出目录（--out_dir）及寄存器宽度（--reg_width）。

完备性：Verilog 模块提供 addr_i、wr_data_i、wr_en_i、rd_data_o 接口，每个寄存器的 _q 和 _we 单独引出。

使用示例

bash
python reg_gen.py registers.xlsx --out_dir ./generated --base_addr 0x4000_0000
