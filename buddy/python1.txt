```python
#!/usr/bin/env python3
"""
IC设计流程自动化脚本
支持Lint、综合、仿真三个步骤的自动化执行
"""

import argparse
import subprocess
import os
import sys
import shutil
from datetime import datetime
from pathlib import Path


def parse_args():
    """解析命令行参数"""
    parser = argparse.ArgumentParser(
        description='IC设计流程自动化脚本',
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
示例:
  python run_flow.py --top top_module --rtl_list rtl.f --syn --sim
  python run_flow.py --top my_top --rtl_list files.lst --syn
        """
    )
    
    parser.add_argument(
        '--top',
        required=True,
        help='顶层模块名'
    )
    
    parser.add_argument(
        '--rtl_list',
        required=True,
        help='RTL文件列表（文本文件，每行一个路径）'
    )
    
    parser.add_argument(
        '--syn',
        action='store_true',
        default=False,
        help='是否执行综合（默认: False）'
    )
    
    parser.add_argument(
        '--sim',
        action='store_true',
        default=False,
        help='是否执行仿真（默认: False）'
    )
    
    return parser.parse_args()


def check_log_for_errors(log_file):
    """
    检查log文件中是否存在ERROR关键字（不区分大小写）
    返回: (bool, list) - (是否存在错误, 错误行列表)
    """
    if not os.path.exists(log_file):
        print(f"警告: log文件不存在: {log_file}")
        return False, []
    
    errors = []
    try:
        with open(log_file, 'r', encoding='utf-8', errors='ignore') as f:
            for line_num, line in enumerate(f, 1):
                if 'ERROR' in line.upper():
                    errors.append((line_num, line.strip()))
    except Exception as e:
        print(f"警告: 读取log文件失败: {e}")
        return False, []
    
    return len(errors) > 0, errors


def move_log_file(log_file, step_name):
    """
    将log文件移动到./logs/目录，并按{步骤名}_{时间戳}.log格式重命名
    """
    # 创建logs目录
    logs_dir = Path('./logs')
    logs_dir.mkdir(exist_ok=True)
    
    # 生成带时间戳的新文件名
    timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
    new_log_name = f"{step_name}_{timestamp}.log"
    new_log_path = logs_dir / new_log_name
    
    # 移动文件
    if os.path.exists(log_file):
        shutil.move(log_file, new_log_path)
        print(f"Log文件已移动: {log_file} -> {new_log_path}")
        return str(new_log_path)
    else:
        print(f"警告: log文件不存在，无法移动: {log_file}")
        return None


def run_lint(top_module, rtl_list):
    """执行Lint检查"""
    print("=" * 60)
    print("开始执行 Lint 检查...")
    print("=" * 60)
    
    log_file = "lint.log"
    
    # 构建命令
    cmd = [
        'spyglass',
        '-run', 'lint',
        '-top', top_module,
        '-filelist', rtl_list,
        '-log', log_file
    ]
    
    print(f"执行命令: {' '.join(cmd)}")
    
    # 执行命令
    try:
        result = subprocess.run(
            cmd,
            check=False,
            text=True
        )
        
        # 检查返回码
        if result.returncode != 0:
            print(f"警告: Spyglass 返回码: {result.returncode}")
        
    except FileNotFoundError:
        print("错误: 未找到 spyglass 命令，请确保已正确设置环境变量")
        return False
    except Exception as e:
        print(f"错误: 执行Lint检查失败: {e}")
        return False
    
    # 检查log文件中的错误
    has_error, error_list = check_log_for_errors(log_file)
    
    if has_error:
        print(f"错误: Lint检查中发现 {len(error_list)} 个ERROR")
        for line_num, line_content in error_list[:10]:  # 只显示前10个错误
            print(f"  L{line_num}: {line_content}")
        if len(error_list) > 10:
            print(f"  ... 还有 {len(error_list) - 10} 个错误未显示")
        
        # 移动log文件
        move_log_file(log_file, "lint")
        return False
    
    print("✓ Lint检查通过")
    
    # 移动log文件
    move_log_file(log_file, "lint")
    return True


def run_synthesis(top_module, rtl_list):
    """执行综合"""
    print("=" * 60)
    print("开始执行综合...")
    print("=" * 60)
    
    log_file = "syn.log"
    
    # 构建命令
    # 注意：dc_shell通常通过Tcl脚本执行，这里需要根据实际情况调整
    cmd = [
        'dc_shell',
        '-f', 'run_synthesis.tcl',
        '-top', top_module,
        '-rtl_list', rtl_list,
        '-log', log_file
    ]
    
    print(f"执行命令: {' '.join(cmd)}")
    
    # 执行命令
    try:
        result = subprocess.run(
            cmd,
            check=False,
            text=True
        )
        
        # 检查返回码
        if result.returncode != 0:
            print(f"警告: dc_shell 返回码: {result.returncode}")
        
    except FileNotFoundError:
        print("错误: 未找到 dc_shell 命令，请确保已正确设置环境变量")
        return False
    except Exception as e:
        print(f"错误: 执行综合失败: {e}")
        return False
    
    # 检查log文件中的错误
    has_error, error_list = check_log_for_errors(log_file)
    
    if has_error:
        print(f"错误: 综合中发现 {len(error_list)} 个ERROR")
        for line_num, line_content in error_list[:10]:  # 只显示前10个错误
            print(f"  L{line_num}: {line_content}")
        if len(error_list) > 10:
            print(f"  ... 还有 {len(error_list) - 10} 个错误未显示")
        
        # 移动log文件
        move_log_file(log_file, "syn")
        return False
    
    print("✓ 综合通过")
    
    # 移动log文件
    move_log_file(log_file, "syn")
    return True


def run_simulation(top_module, rtl_list):
    """执行仿真"""
    print("=" * 60)
    print("开始执行仿真...")
    print("=" * 60)
    
    log_file = "sim.log"
    
    # 构建命令
    cmd = [
        'vcs',
        '-f', rtl_list,
        '-top', top_module,
        '+define+SIM',
        '-l', log_file
    ]
    
    print(f"执行命令: {' '.join(cmd)}")
    
    # 执行命令
    try:
        result = subprocess.run(
            cmd,
            check=False,
            text=True
        )
        
        # 检查返回码
        if result.returncode != 0:
            print(f"警告: VCS 返回码: {result.returncode}")
        
    except FileNotFoundError:
        print("错误: 未找到 vcs 命令，请确保已正确设置环境变量")
        return False
    except Exception as e:
        print(f"错误: 执行仿真失败: {e}")
        return False
    
    # 检查log文件中的错误
    has_error, error_list = check_log_for_errors(log_file)
    
    if has_error:
        print(f"错误: 仿真中发现 {len(error_list)} 个ERROR")
        for line_num, line_content in error_list[:10]:  # 只显示前10个错误
            print(f"  L{line_num}: {line_content}")
        if len(error_list) > 10:
            print(f"  ... 还有 {len(error_list) - 10} 个错误未显示")
        
        # 移动log文件
        move_log_file(log_file, "sim")
        return False
    
    print("✓ 仿真通过")
    
    # 移动log文件
    move_log_file(log_file, "sim")
    return True


def main():
    """主函数"""
    # 解析命令行参数
    args = parse_args()
    
    print("=" * 60)
    print("IC设计流程自动化脚本")
    print("=" * 60)
    print(f"顶层模块: {args.top}")
    print(f"RTL文件列表: {args.rtl_list}")
    print(f"执行综合: {args.syn}")
    print(f"执行仿真: {args.sim}")
    print("=" * 60)
    
    # 检查RTL文件列表是否存在
    if not os.path.exists(args.rtl_list):
        print(f"错误: RTL文件列表不存在: {args.rtl_list}")
        sys.exit(1)
    
    # 执行流程
    # 1. Lint检查（总是执行）
    if not run_lint(args.top, args.rtl_list):
        print("错误: Lint检查失败，停止后续步骤")
        sys.exit(1)
    
    # 2. 综合（如果启用）
    if args.syn:
        if not run_synthesis(args.top, args.rtl_list):
            print("错误: 综合失败，停止后续步骤")
            sys.exit(1)
    
    # 3. 仿真（如果启用）
    if args.sim:
        if not run_simulation(args.top, args.rtl_list):
            print("错误: 仿真失败")
            sys.exit(1)
    
    print("=" * 60)
    print("✓ 所有流程执行成功")
    print("=" * 60)
    sys.exit(0)


if __name__ == '__main__':
    main()
```

## 脚本特性说明

### 1. **参数处理**
- 使用 `argparse` 库进行专业的命令行参数解析
- `--top` 和 `--rtl_list` 是必需参数
- `--syn` 和 `--sim` 是可选标志参数，默认为 `False`

### 2. **执行流程**
- **Lint检查**：总是执行（这是IC设计的标准流程）
- **综合**：仅当 `--syn` 启用时执行
- **仿真**：仅当 `--sim` 启用时执行

### 3. **错误处理**
- 每个步骤完成后，自动检查log文件中的 `ERROR` 关键字（不区分大小写）
- 如果发现错误，立即停止后续步骤，并返回退出码 1
- 如果所有步骤成功，返回退出码 0

### 4. **日志管理**
- 所有log文件自动移动到 `./logs/` 目录
- 按 `{步骤名}_{时间戳}.log` 格式重命名
- 时间戳格式：`YYYYMMDD_HHMMSS`（例如：`lint_20260430_112130.log`）

### 5. **其他特性**
- 彩色输出（使用Unicode符号 ✓ 和 分隔线）
- 详细的错误报告（显示前10个错误）
- 文件存在性检查
- 命令执行异常处理
