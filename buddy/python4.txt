def detect_tool() -> Tuple[Optional[str], Optional[str]]:
    """
    自动检测可用的覆盖率工具。
    返回 (tool_name, tool_path)，优先级: urg > imc。
    """
    candidates = [
        ("urg", ["urg", "-version"]),
        ("imc", ["imc", "-version"]),
    ]
    for tool_name, version_cmd in candidates:
        try:
            result = subprocess.run(
                version_cmd,
                capture_output=True, text=True, timeout=10
            )
            # urg 在 stdout，imc 在 stderr，均视为成功
            if result.returncode == 0 or "version" in (result.stdout + result.stderr).lower():
                tool_path = shutil.which(tool_name)
                print(f"  ✅ 检测到工具: {tool_name} ({tool_path})")
                return tool_name, tool_path
        except (FileNotFoundError, subprocess.TimeoutExpired):
            continue

    return None, None

    """

    if not search_path.is_dir():
        raise NotADirectoryError(f"搜索目录不存在: {search_dir}")

    databases: List[Path] = []

    if pattern == "*.vdb":
        # Synopsys: *.vdb 是目录

            if p.is_dir():
                databases.append(p)
    else:
        # Cadence 或其他格式
        for p in search_path.rglob(pattern):
            if p.is_dir():
                databases.append(p)

    return sorted(databases)

    databases: List[Path],

    extra_opts: List[str] = None,
    norecurse: bool = False

    命令: urg -full64 -dir <db1> <db2> ... -dbname <output_db> [options]
    """
    cmd = ["urg", "-full64", "-dir"] + [str(db) for db in databases]
    cmd += ["-dbname", str(output_db)]

    if norecurse:
        cmd.append("-norecurse")

    if extra_opts:
        cmd.extend(extra_opts)

    print(f"  ▶ 执行: {' '.join(str(c) for c in cmd)}")

    result = subprocess.run(
        cmd,
        capture_output=True, text=True,
        cwd=output_db.parent if output_db.parent.exists() else None
    )

    if result.returncode != 0:
        print(f"  ❌ urg 合并失败!")
        print(f"  stderr: {result.stderr[-2000:]}")   # 只打印最后 2000 字符
        return False

    print(f"  ✅ urg 合并完成: {output_db}")
    return True

    print(f"  ✅ imc 报告已生成: {csv_output}")
    return True


# ========== 解析 urg XML 报告 ==========
def parse_urg_xml_report(report_dir: Path, top_module: str) -> List[Dict]:
    """
    解析 urg 生成的 XML 报告，提取每个模块的行覆盖率和条件覆盖率。
    urg -report xml 会生成 coverage_results.xml。
    """
    results = []

    # urg XML 报告路径（常见位置）
    xml_candidates = [
        report_dir / "coverage_results.xml",
        report_dir / "urgReport" / "coverage_results.xml",
        report_dir / "xml" / "coverage_results.xml",
    ]

    xml_file = None
    for candidate in xml_candidates:
        if candidate.exists():
            xml_file = candidate
            break

    if xml_file is None:
        print(f"  ⚠️  未找到 urg XML 报告，尝试解析文本报告...")
        return parse_urg_text_report(report_dir, top_module)

    print(f"  📖 解析 urg XML 报告: {xml_file}")

    try:
        tree = ET.parse(xml_file)
        root = tree.getroot()

        # urg XML 结构（示意）:
        # <coverage>
        #   <module name="top" line_cov="95.2" cond_cov="87.5"/>
        # </coverage>
        for module_elem in root.iter("module"):
            module_name = module_elem.get("name", "")
            if module_name == top_module or module_name.startswith(top_module + "."):
                line_cov = float(module_elem.get("line_cov", 0.0))
                cond_cov = float(module_elem.get("cond_cov", 0.0))
                results.append({
                    "module_name":        module_name,
                    "line_coverage":      line_cov,
                    "condition_coverage": cond_cov,
                })

    except ET.ParseError as e:
        print(f"  ⚠️  XML 解析失败: {e}，尝试文本报告...")
        return parse_urg_text_report(report_dir, top_module)

    # 按模块名排序
    results.sort(key=lambda x: x["module_name"])
    return results


# ========== 解析 urg 文本报告 ==========
def parse_urg_text_report(report_dir: Path, top_module: str) -> List[Dict]:
    """
    解析 urg 生成的文本报告（urgReport/coverage_summary.rpt 或类似文件）。
    文本格式示例:
        Module            Line     Condition
        top               95.2%    87.5%
        top.axi_master    98.1%    91.2%
    """
    results = []

    # 常见文本报告路径
    rpt_candidates = [
        report_dir / "coverage_summary.rpt",
        report_dir / "urgReport" / "coverage_summary.rpt",
        report_dir / "urgReport" / "summary.rpt",
        report_dir / "summary.rpt",
    ]

    rpt_file = None
    for candidate in rpt_candidates:
        if candidate.exists():
            rpt_file = candidate
            break

    if rpt_file is None:
        print(f"  ❌ 未找到 urg 文本报告文件")
        return results

    print(f"  📖 解析 urg 文本报告: {rpt_file}")

    content = rpt_file.read_text(encoding="utf-8", errors="replace")

    # 正则：匹配 "Module" 表头后的数据行
    # 模块名（可含点号）  行覆盖率%  条件覆盖率%
    # 例如: top.axi_master  98.1%  91.2%
    pattern = re.compile(
        r'^(\S+(?:\.\S+)*)'          # module name (支持 top.sub1.sub2)
        r'\s+'                        # 空白
        r'([\d.]+)%?'                # line coverage
        r'\s+'                        # 空白
        r'([\d.]+)%?',               # condition coverage
        re.MULTILINE
    )

    in_table = False
    for line in content.splitlines():
        line_stripped = line.strip()

        # 检测表头开始
        if re.search(r'Module.*Line.*Condition', line_stripped, re.IGNORECASE):
            in_table = True
            continue

        if not in_table:
            continue

        # 检测表尾（空行或分隔线）
        if not line_stripped or re.match(r'^[-=]+$', line_stripped):
            continue

        m = pattern.match(line_stripped)
        if m:
            module_name = m.group(1)
            if module_name == top_module or module_name.startswith(top_module + "."):
                line_cov = float(m.group(2))
                cond_cov = float(m.group(3))
                results.append({
                    "module_name":        module_name,
                    "line_coverage":      line_cov,
                    "condition_coverage": cond_cov,
                })

    results.sort(key=lambda x: x["module_name"])
    return results


# ========== 解析 imc CSV 报告 ==========
def parse_imc_csv_report(report_dir: Path, top_module: str) -> List[Dict]:
    """
    解析 imc 生成的 CSV 覆盖率报告。
    """
    csv_file = report_dir / IMC_REPORT_FILE

    if not csv_file.exists():
        print(f"  ⚠️  未找到 imc CSV 报告: {csv_file}")
        return []

    print(f"  📖 解析 imc CSV 报告: {csv_file}")

    results = []
    with open(csv_file, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for row in reader:
            # imc CSV 列名可能不同，尝试常见命名
            module_name = (
                row.get("Module") or row.get("module") or
                row.get("Hierarchy") or row.get("hierarchy") or ""
            ).strip()

            if not (module_name == top_module or module_name.startswith(top_module + ".")):
                continue

            # 提取覆盖率数值（去掉 % 符号）
            line_cov_str = (
                row.get("Line") or row.get("line_coverage") or
                row.get("Line Coverage") or "0"
            ).strip().rstrip('%')

            cond_cov_str = (
                row.get("Condition") or row.get("condition_coverage") or
                row.get("Cond Coverage") or "0"
            ).strip().rstrip('%')

            try:
                line_cov = float(line_cov_str)
                cond_cov = float(cond_cov_str)
            except ValueError:
                continue

            results.append({
                "module_name":        module_name,
                "line_coverage":      line_cov,
                "condition_coverage": cond_cov,
            })

    results.sort(key=lambda x: x["module_name"])
    return results


# ========== 写入 CSV 报告 ==========
def write_csv_report(modules: List[Dict], output_csv: Path) -> None:
    """
    将覆盖率数据写入 CSV 文件。
    格式:
        Module Name, Line Coverage, Condition Coverage
        top, 95.2%, 87.5%
    """
    output_csv.parent.mkdir(parents=True, exist_ok=True)

    with open(output_csv, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(["Module Name", "Line Coverage", "Condition Coverage"])

        for m in modules:
            writer.writerow([
                m["module_name"],
                f"{m['line_coverage']:.1f}%",
                f"{m['condition_coverage']:.1f}%",
            ])

    print(f"\n  ✅ CSV 报告已写入: {output_csv}")


# ========== 覆盖率阈值检查与警告 ==========
def check_and_warn(modules: List[Dict]) -> None:
    """
    检查覆盖率是否低于阈值，打印警告信息。
    """
    cond_warnings = []
    line_warnings = []

    for m in modules:
        if m["condition_coverage"] < COND_COVERAGE_WARN_THRESHOLD:
            cond_warnings.append(
                f"  ⚠️  {m['module_name']:<30s} "
                f"条件覆盖率 = {m['condition_coverage']:.1f}% "
                f"(阈值 {COND_COVERAGE_WARN_THRESHOLD:.0f}%)"
            )
        if m["line_coverage"] < LINE_COVERAGE_WARN_THRESHOLD:
            line_warnings.append(
                f"  ⚠️  {m['module_name']:<30s} "
                f"行覆盖率 = {m['line_coverage']:.1f}% "
                f"(阈值 {LINE_COVERAGE_WARN_THRESHOLD:.0f}%)"
            )

    if cond_warnings or line_warnings:
        print("\n" + "=" * 70)
        print("  覆盖率警告")
        print("=" * 70)
        for w in cond_warnings + line_warnings:
            print(w)
    else:
        print(f"\n  ✅ 所有模块覆盖率均满足阈值要求 "
              f"(行≥{LINE_COVERAGE_WARN_THRESHOLD:.0f}%, "
              f"条件≥{COND_COVERAGE_WARN_THRESHOLD:.0f}%)")


# ========== 打印摘要 ==========
def print_summary(modules: List[Dict], top_module: str) -> None:
    """打印覆盖率摘要表格。"""
    print("\n" + "=" * 70)
    print(f"  覆盖率报告 - 顶层模块: {top_module}")
    print("=" * 70)
    print(f"  {'Module Name':<30s} {'Line':>10s} {'Condition':>12s}")
    print("  " + "-" * 66)

    for m in modules:
        line_flag = " ⚠️" if m["line_coverage"] < LINE_COVERAGE_WARN_THRESHOLD else ""
        cond_flag = " ⚠️" if m["condition_coverage"] < COND_COVERAGE_WARN_THRESHOLD else ""
        print(f"  {m['module_name']:<30s} "
              f"{m['line_coverage']:>9.1f}%{line_flag} "
              f"{m['condition_coverage']:>11.1f}%{cond_flag}")

    print("=" * 70)


# ========== 主流程 ==========
def main():
    parser = argparse.ArgumentParser(
        description="合并覆盖率数据库并生成 CSV 报告（支持 urg / imc）",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
示例:
  # 自动检测工具，合并当前目录下所有 *.vdb
  python3 %(prog)s --search-dir ./cov_work/scope

  # 指定使用 urg，并添加 -norecurse 选项
  python3 %(prog)s --tool urg --norecurse --extra-opts "-metric line+cond"

  # 指定顶层模块名
  python3 %(prog)s --top-module my_dut

  # 调整警告阈值
  python3 %(prog)s --cond-threshold 85 --line-threshold 75
"""
    )

    parser.add_argument(
        "--search-dir", default=".",
        help="扫描覆盖率数据库的目录 (默认: 当前目录)"
    )
    parser.add_argument(
        "--db-pattern", default="*.vdb",
        help="覆盖率数据库匹配模式 (默认: *.vdb)"
    )
    parser.add_argument(
        "--output-db", default="merged.vdb",
        help="合并后数据库名称 (默认: merged.vdb)"
    )
    parser.add_argument(
        "--report-dir", default="coverage_report",
        help="报告输出目录 (默认: coverage_report)"
    )
    parser.add_argument(
        "--output-csv", default="coverage_report.csv",
        help="CSV 报告文件名 (默认: coverage_report.csv)"
    )
    parser.add_argument(
        "--top-module", default="top",
        help="顶层模块名，用于过滤子模块 (默认: top)"
    )
    parser.add_argument(
        "--tool", choices=["urg", "imc", "auto"], default="auto",
        help="覆盖率工具 (默认: auto 自动检测)"
    )
    parser.add_argument(
        "--norecurse", action="store_true",
        help="合并时添加 -norecurse 选项（urg）"
    )
    parser.add_argument(
        "--extra-opts", nargs="*", default=[],
        help="传递给合并命令的额外选项"
    )
    parser.add_argument(
        "--cond-threshold", type=float, default=COND_COVERAGE_WARN_THRESHOLD,
        help=f"条件覆盖率警告阈值 %% (默认: {COND_COVERAGE_WARN_THRESHOLD})"
    )
    parser.add_argument(
        "--line-threshold", type=float, default=LINE_COVERAGE_WARN_THRESHOLD,
        help=f"行覆盖率警告阈值 %% (默认: {LINE_COVERAGE_WARN_THRESHOLD})"
    )
    parser.add_argument(
        "--skip-merge", action="store_true",
        help="跳过合并步骤，直接解析已有报告"
    )

    args = parser.parse_args()

    # 更新全局阈值
    global COND_COVERAGE_WARN_THRESHOLD, LINE_COVERAGE_WARN_THRESHOLD
    COND_COVERAGE_WARN_THRESHOLD = args.cond_threshold
    LINE_COVERAGE_WARN_THRESHOLD  = args.line_threshold

    print("=" * 70)
    print("  覆盖率合并与报告生成脚本")
    print("=" * 70)

    # ---- Step 1: 检测工具 ----
    print("\n[Step 1] 检测覆盖率工具...")
    if args.tool == "auto":
        tool, _ = detect_tool()
        if tool is None:
            print("  ❌ 未找到 urg 或 imc，请确认工具已正确安装并在 PATH 中")
            sys.exit(1)
    else:
        tool = args.tool
        print(f"  ✅ 使用指定工具: {tool}")

    # ---- Step 2: 扫描覆盖率数据库 ----
    print(f"\n[Step 2] 扫描覆盖率数据库: {args.search_dir}/{args.db_pattern}")
    databases = find_coverage_databases(args.search_dir, args.db_pattern)

    if not databases:
        print(f"  ❌ 未找到任何覆盖率数据库（匹配模式: {args.db_pattern}）")
        sys.exit(1)

    print(f"  ✅ 找到 {len(databases)} 个覆盖率数据库:")
    for db in databases:
        print(f"     - {db}")

    # ---- Step 3: 合并数据库 ----
    output_db  = Path(args.output_db).resolve()
    report_dir = Path(args.report_dir).resolve()
    output_csv = Path(args.output_csv).resolve()

    if not args.skip_merge:
        print(f"\n[Step 3] 合并覆盖率数据库 -> {output_db}")
        if tool == "urg":
            success = merge_with_urg(databases, output_db, args.extra_opts, args.norecurse)
        else:
            success = merge_with_imc(databases, output_db, args.extra_opts)

        if not success:
            print("  ❌ 合并失败，退出")
            sys.exit(1)

        # ---- Step 4: 生成报告 ----
        print(f"\n[Step 4] 生成覆盖率报告...")
        if tool == "urg":
            generate_urg_report(output_db, report_dir)
        else:
            generate_imc_report(output_db, report_dir)
    else:
        print("\n[Step 3/4] 跳过合并，直接解析已有报告...")

    # ---- Step 5: 解析报告 ----
    print(f"\n[Step 5] 解析覆盖率报告...")
    if tool == "urg":
        modules = parse_urg_xml_report(report_dir, args.top_module)
        if not modules:
            modules = parse_urg_text_report(report_dir, args.top_module)
    else:
        modules = parse_imc_csv_report(report_dir, args.top_module)

    if not modules:
        print("  ❌ 未能从报告中解析出任何模块数据")
        print("  请检查报告目录内容，或手动指定 --report-dir")
        sys.exit(1)

    print(f"  ✅ 解析到 {len(modules)} 个模块")

    # ---- Step 6: 写入 CSV ----
    print(f"\n[Step 6] 写入 CSV 报告...")
    write_csv_report(modules, output_csv)

    # ---- Step 7: 打印摘要 & 警告 ----
    print_summary(modules, args.top_module)
    check_and_warn(modules)

    print(f"\n✅ 全部完成！CSV 报告: {output_csv}")
    print("=" * 70)


if __name__ == "__main__":
    main()
